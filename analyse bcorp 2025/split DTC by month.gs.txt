function splitDTCByMonth() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  var src = ss.getSheetByName('DTC');
  if (!src) throw new Error('No tab named DTC found.');

  var data = src.getDataRange().getValues();
  if (data.length < 2) throw new Error('DTC tab has no data rows.');

  var header = data[0];
  var width = header.length;

  var dateCol = -1;
  for (var c = 0; c < width; c++) {
    if (String(header[c]).trim().toLowerCase() === 'date') {
      dateCol = c;
      break;
    }
  }
  if (dateCol === -1) throw new Error('Could not find a date column in the DTC header row.');

  var buckets = {};
  var unsorted = [];

  for (var r = 1; r < data.length; r++) {
    var row = data[r];

    var blank = true;
    for (var k = 0; k < row.length; k++) {
      if (row[k] !== '' && row[k] !== null) {
        blank = false;
        break;
      }
    }
    if (blank) continue;

    var d = row[dateCol];
    var month = null;
    if (Object.prototype.toString.call(d) === '[object Date]' && !isNaN(d.getTime())) {
      month = d.getMonth() + 1;
    } else if (d !== '' && d !== null) {
      var p = new Date(d);
      if (!isNaN(p.getTime())) month = p.getMonth() + 1;
    }
    if (month === null) {
      unsorted.push(row);
      continue;
    }

    var key = ('0' + month).slice(-2);
    if (!buckets[key]) buckets[key] = [];
    buckets[key].push(row);
  }

  var summary = [];

  for (var m = 1; m <= 12; m++) {
    var mkey = ('0' + m).slice(-2);
    if (!buckets[mkey]) continue;

    var rows = buckets[mkey];
    var sheet = ss.getSheetByName(mkey);
    if (sheet) {
      sheet.clear();
    } else {
      sheet = ss.insertSheet(mkey);
    }

    var out = [header].concat(rows);
    sheet.getRange(1, 1, out.length, width).setValues(out);
    sheet.setFrozenRows(1);
    summary.push(mkey + '=' + rows.length);
  }

  if (unsorted.length > 0) {
    var us = ss.getSheetByName('DTC_no_date');
    if (us) {
      us.clear();
    } else {
      us = ss.insertSheet('DTC_no_date');
    }
    var uo = [header].concat(unsorted);
    us.getRange(1, 1, uo.length, width).setValues(uo);
    summary.push('no_date=' + unsorted.length);
  }

  ss.toast('DTC split done: ' + summary.join('  '), 'Complete', 8);
  Logger.log(summary.join('  '));
}
